export type RouteResult = {
    route: string;
    verdict: Verdict;
    slope: string;
    /** Heap after a forced GC, once per cycle, in MB. */
    samples: number[];
    /** The object at the head of the retainer chain, as the table cell shows it. */
    holder?: string;
    retainer?: string;
};
